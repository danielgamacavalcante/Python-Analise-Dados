  WHEN DSC_ITEM_ELEMENTO = 'CONSTITUICAO OU AUMENTO DE CAPITAL DE EM' THEN 'CONSTITUICAO OU AUMENTO DE CAPITAL DE EMPRESAS'
  WHEN DSC_ITEM_ELEMENTO = 'CONTRIBUICAO A ENTIDADES FECHADAS DE PRE' THEN 'CONTRIBUICAO A ENTIDADES FECHADAS DE PREVIDENCIA'
  WHEN DSC_ITEM_ELEMENTO = 'OUTRAS DESPESAS DE PESSOAL DECORRENTES D' THEN 'OUTRAS DESPESAS DE PESSOAL DECORRENTES DE CONTRATOS DE TERCEIRIZACAO'
  WHEN DSC_ITEM_ELEMENTO = 'OUTRAS DESPESAS VARIAVEIS - PESSOAL CIVI' THEN 'OUTRAS DESPESAS VARIAVEIS - PESSOAL CIVIL'
  WHEN DSC_ITEM_ELEMENTO = 'OUTROS AUXILIOS FINANCEIROS A PESSOAS FI' THEN 'OUTROS AUXILIOS FINANCEIROS A PESSOAS FISICAS'

  WHEN DSC_ITEM_ELEMENTO = 'OUTROS SERVICOS DE TERCEIROS - PESSOA JU' THEN 'OUTROS SERVICOS DE TERCEIROS - PESSOA JURIDICA'
  WHEN DSC_ITEM_ELEMENTO = 'PENSOES ESPECIAIS.' THEN 'PENSOES ESPECIAIS'
  WHEN DSC_ITEM_ELEMENTO = 'PREMIACOES CULTURAIS, ARTISTICAS, CIENTI' THEN 'PREMIACOES CULTURAIS, ARTISTICAS, CIENTIFICAS, DESPORTIVAS E OUTRAS'

  WHEN DSC_ITEM_ELEMENTO = 'RESSARCIMENTO DE DESPESAS DE PESSOAL REQ' THEN 'RESSARCIMENTO DE DESPESAS DE PESSOAL REQUISITADO'
  WHEN DSC_ITEM_ELEMENTO = 'SERVICO DE CONSULTORIA' THEN 'SERVICOS DE CONSULTORIA'

	D_FUNC.ID AS ID_FUNCAO,

	D_SUBFUNC.ID AS ID_SUBFUNCAO,
	D_CATEG.ID AS ID_CATEGORIA,
	D_MODALI.ID AS ID_MODALIDADE,
	D_ORG.ID_ORGAO AS ID_ORGAO,
	DT_EMP.ID AS ID_DAT_EMPENHO,

INNER JOIN DW_EFD.DIM_PROGRAMA D_PROG ON D_PROG.DSC_PROGRAMA = EFD.DSC_PROGRAMA
INNER JOIN DW_EFD.DIM_SUBFUNCAO D_SUBFUNC ON D_SUBFUNC.DSC_SUBFUNCAO = EFD.DSC_SUBFUNCAO
