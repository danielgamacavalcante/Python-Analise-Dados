Faturamento Bruto = A quantidade de produtos * o preço unitário por produto

Receita Bruta (É o valor Total dos produtos - menos o desconto de venda)

Desconto (Pode ser calculado do Faturamento Bruto - a Receita Bruta)

Receita líquida (É o valor da Receita Bruta - Impostos(tributos))

Lucro (Margem)(É a receita líquida menos - os custos(De pessoal,matéria prima, ambiente de produção, estoque etc))
==========================================
00.Medidas Diversas: 

Contagem de Linhas Fato_vendas =
	COUNTROWS(fato_vendas)

Quantidade Produto Vendida = SUM(fato_vendas[Qtde])

Média Qtde Vendida =
	AVERAGE(fato_vendas[Qtde])

Qtde Máxima Vendida =
	MAX(fato_vendas[Qtde])

Qtde Mínima Vendida =
	MIN(fato_vendas[Qtde])

01. Performance
Faturamento Bruto =
	SUMX(fato_vendas, fato_vendas[Qtde] * RELATED(dim_produto[Preco_Unitario]))

Receita Bruta com Desconto = SUM(fato_vendas[Valor Total])

Desconto = [Faturamento Bruto] - [Receita Bruta]

Custos =
	SUMX(fato_vendas, fato_vendas[Qtde] * RELATED(dim_produto[Custo])

Tributos = 
	SUMX(fato_vendas, fato_vendas[Qtde] * RELATED(dim_produto[Tributos])

Receita Líquida = 
	[Receita Bruta] - [Tributos]

Margem = [Receita Líquida] - [Custos]

% Margem =
	DIVIDE([Margem], [Receita Líquida]), 0)

02.Calculate

ALL - Puxa tudo da tabela e não respeita filtro.

Faturamento Bruto - ALL =
	CALCULATE(
		[Faturamento Bruto],
		ALL(dim_categoria[Categoria])
 

% Fat Bruto Total - All = 
	DIVIDE([Faturamento Bruto], [Faturamento Bruto - ALL],0)

ALLSELECTED - Puxa tudo da tabela e respeita a utilização de filtro.

Faturamento Bruto - ALLSELECTED = 
	CALCULATE(
		[Faturamento Bruto],
		ALLSELECTED(dim_categoria[Categoria])

% Fat Bruto Total - ALLSELECTED =
	DIVIDE([Faturamento Bruto], [Faturamento Bruto - ALLSELECTED],0)

FILTER:

    [Faturamento Bruto],

        ALLSELECTED(dim_calendario),
        dim_calendario[Data] <= MAX(fato_vendas[Data_Venda])
 ))

03.Inteligência de Tempo

Faturamento Bruto LY =
	CALCULATE(
		[Faturamento Bruto],
		SAMEPERIODLASTYEAR(dim_calendario[Data])

Faturamento Bruto LM =
	CALCULATE(
		[Faturamento Bruto],
		DATEADD(
			dim_calendario[Data],
			-1,
			MONTH)
% YoY Faturamento Bruto =
	DIVIDE(
	[Faturamento Bruto] - [Faturamento Bruto LY],
	[Faturamento Bruto LY],
	0)

05.ISINSCOPE(Parametro1,Parametro2) (Retorna True ou False, no primeiro parametro. Se for true, ele executa o que se pede depois):

% sobre o subtotal Receita Bruta = 
    var TotalProduto = CALCULATE([Receita Bruta],ALLSELECTED(dim_produto[Descricao_Produto]))
    var TotalSubcategoria = CALCULATE([Receita Bruta], ALLSELECTED(dim_subcategoria[Subcategoria]))
    var TotalCategoria = CALCULATE([Receita Bruta],ALLSELECTED(dim_categoria[Categoria]))
RETURN
SWITCH(
    TRUE(),
    ISINSCOPE(dim_produto[Descricao_Produto]),DIVIDE([Receita Bruta],TotalProduto,0),
    ISINSCOPE(dim_subcategoria[Subcategoria]),DIVIDE([Receita Bruta],TotalSubcategoria,0),
    ISINSCOPE(dim_categoria[Categoria]), DIVIDE([Receita Bruta],TotalCategoria,0)
    ,BLANK())
